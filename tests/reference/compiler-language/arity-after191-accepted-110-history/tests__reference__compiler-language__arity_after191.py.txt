def rows():
    if digest(CATALOG.read_bytes()) != CATALOG_SHA256 or digest((ROOT/SCANNER).read_bytes()) != SCANNER_SHA256:
        raise ValueError('frozen catalog/scanner drift')
    driver = (ROOT/DRIVER).read_text()
    for required in [TEST, 'MacroHost::new()', 'compile_namespace_with_macros', "list 'quote (do {source})", 'fresh', 'reused', 'arity-after191-observation', 'sci_execution_observation.js']:
        if required not in driver: raise ValueError('actual source phase driver detached')
    original_rows = {line.split('\t')[0]:line for line in (ROOT/'tests/fixtures/execution-outcomes.tsv').read_text().splitlines() if line and not line.startswith('#')}
    result=[]
    for row in REVIEWED:
        original = original_rows[row['original_id']]
        fields=original.split('\t')
        case={'id':row['id'],'source':fields[1],'expected':{'id':row['id'],'outcome':json.loads(fields[2]),'trace':json.loads(fields[3]),'stdout':json.loads(fields[4])},'native_stdout':None}
        if digest(original)!=row['original_row_sha256'] or (ROOT/row['fixture']).read_text()!=fields[1] or digest(fields[1])!=row['source_sha256'] or case!=row['case'] or digest(json.dumps(case,sort_keys=True))!=row['case_sha256']:
            raise ValueError('whole original source/result/effects drift')
        if json.dumps(row['id']) not in driver or Path(row['fixture']).name not in driver:
            raise ValueError('whole original fixture omitted by Rust driver')
        result.append({**row,'source':fields[1]})
    return result


def supplemental_reference_cases():
    return {row['id']:row['case'] for row in rows()}


def demand_basis(row):
    return DEMAND[row['original_id']]


def operation_evidence(source, selected, target):
    if target['name'] != selected['call_name']:
        raise ValueError('operation spelling drift')
    calls=[call for call in direct_calls(source, target['name']) if call['argument_count'] in selected['direct_arities']]
    if {call['argument_count'] for call in calls}!=set(selected['direct_arities']):
        raise ValueError('exact executing direct call omitted')
    signature=target.get('signature')
    if signature and not all(call['argument_count']==signature['minimum'] if signature['kind']=='fixed' else call['argument_count']>signature['minimum'] for call in calls):
        raise ValueError('exact signature demand omitted')
    return calls

def audit_installation(exported):
    """Accept no installed cohort, or every exact own paired row; reject drift."""
    owned={row['obligation_id'] for row in exported['obligation-witnesses.jsonl']}
    installed={}
    for name in ['obligation-witnesses.jsonl','phase-witnesses.jsonl']:
        expected={row['obligation_id']:row for row in exported[name]}
        rows=[row for row in map(json.loads,(HERE/name).read_text().splitlines()) if row['obligation_id'] in owned]
        ids=[row['obligation_id'] for row in rows]
        if len(ids)!=len(set(ids)):
            raise ValueError('duplicate own installed loader row: '+name)
        if any(row!=expected[row['obligation_id']] for row in rows):
            raise ValueError('own installed loader overlap or source/contract/execution drift: '+name)
        installed[name]=set(ids)
    if any(installed.values()) and any(ids!=owned for ids in installed.values()):
        raise ValueError('partial or unpaired own cohort installation')
    return {'state':'installed-exact-paired' if any(installed.values()) else 'not-installed',
        'ordinary_rows':len(installed['obligation-witnesses.jsonl']),
        'phase_rows':len(installed['phase-witnesses.jsonl']),
        'expected_owned_rows':len(owned),
        'overlap_or_drift':'rejected by exact whole-row comparison'}


def audit_installed_loaders(exported):
    state=audit_installation(exported)
    if state['state']=='not-installed':
        return {**state,'ordinary_loader':'pending-integration','phase_loader':'pending-integration'}
    import full_ledger,report
    assertions,obligations,_=full_ledger.load()
    _,witnesses,_=report.load_contract('intended-language')
    installed={
        'obligation-witnesses.jsonl':full_ledger.load_obligation_witnesses(obligations,assertions,witnesses),
        'phase-witnesses.jsonl':full_ledger.load_phase_witnesses(obligations)}
    for name in installed:
        actual=[installed[name].get(row['obligation_id']) for row in exported[name]]
        if actual!=exported[name]:raise ValueError('installed permanent loader source/contract/execution drift: '+name)
    return {**state,'ordinary_loader':'passed-exact-own-cohort','phase_loader':'passed-exact-own-cohort'}


def parse_frames(output):
    expected=[(row['id'],host,phase,engine) for row in rows() for host in ['fresh','reused'] for phase in ['application','macro'] for engine in ['QuickJS','Node']]
    contracts={row['id']:row['case']['expected'] for row in rows()};frames=[]
    for line in output.splitlines():
        if not line.startswith('arity-after191-observation\t'):continue
        fields=line.split('\t')
        if len(fields)!=6 or json.loads(fields[5])!=contracts.get(fields[1]):raise ValueError('nonpassing complete actual typed packet')
        frames.append({'id':fields[1],'host':fields[2],'phase':fields[3],'engine':fields[4],'raw':fields[5]})
    if [(f['id'],f['host'],f['phase'],f['engine']) for f in frames]!=expected:raise ValueError('missing/extra/reordered actual frames')
    return frames
