PROFILE='exact-original-sci-originals-after191-v1'
CONTRACTS=Path(__file__).with_name('sci-originals-after191-contracts.json')
CONTRACTS_SHA256='ba738e7e8e0a398b7ba88a07693e9e84ac9cbdd90559ea9f3243abb18e308522'


def load():
    proof.same(sci.digest(CONTRACTS.read_bytes()),CONTRACTS_SHA256,'Unreviewed whole-source complete literal contracts')
    contracts=json.loads(CONTRACTS.read_text());originals=candidates.load()
    proof.same(tuple(r['sci_id'] for r in contracts),candidates.IDS,'Whole-source canonical source cohort changed')
    result=[]
    for contract,row in zip(contracts,originals):
        for key in ('sci_id','body_sha256','original_row_sha256','source_sha256'):
            proof.same(contract[key],row[key],'Whole-source original complete source binding drift')
        packet=contract['canonical_expected']
        proof.same(set(packet),{'id','outcome','trace','stdout'},'Incomplete whole-source canonical packet')
        proof.same(packet['id'],row['id'],'Detached whole-source canonical source identity')
        if row['sci_id']=='sci-xf-bounds-empty-invalid':
            proof.same(packet['outcome'],{'status':'threw','error':{'category':'error'}},'Changed full canonical original structured error')
        else:
            proof.same(packet['outcome']['status'],'returned','Whole-source canonical return contract changed')
        if sci.report_module().same_json(packet,row['expected']):raise ValueError('Whole-original canonical difference lost')
        proof.same(packet['stdout'],row['expected']['stdout'],'Whole-source original stdout differs')
        if not contract['reason'] or not contract['scope_basis']:raise ValueError('Missing whole-source canonical scope basis')
        result.append({**row,'original_expected':copy.deepcopy(row['expected']),
          'canonical_expected':copy.deepcopy(packet),'reason':contract['reason'],
          'scope_basis':contract['scope_basis'],'focused_owner':None,
          'source_context_profile':PROFILE,'original_semantic_pass_credit':0,
          'original_expectation_status':'unresolved'})
    return result


def snapshot():
    paths=[Path(__file__),Path(executable.__file__),Path(pin_inputs.__file__),sci.HERE/'full_ledger.py']
    return {**proof.snapshot(),**candidates.snapshot(),
            **{str(p.relative_to(sci.ROOT)):sci.digest(p.read_bytes()) for p in paths}}


def node_script(folder,index):
    return ('process.chdir('+json.dumps(str(folder))+');\n'
      +'require('+json.dumps(str(folder/'reference.js'))+');\n'
      +'const selected = '+json.dumps([index])+';\n'
      +'globalThis.lightCases = selected.map(i => lightCases[i]);\n'
      +'globalThis.lightCaseMetadata = selected.map(i => lightCaseMetadata[i]);\n'
      +(sci.ROOT/'tests/support/execution.js').read_text())


def compiled_pin_units(root):
    return {str(p.relative_to(root)):sci.digest(p.read_bytes()) for p in sorted(root.rglob('*.js')) if p.is_file()}


def source_tree(root):
    return {str(p.relative_to(root)):p.read_bytes() for p in sorted(root.rglob('*')) if p.is_file()}


def location_command(execution,output,kind,programs):
    command=json.loads((execution/'build/invocation.json').read_text())
    identity=sci.report_module().RELEASE if kind=='release' else sci.report_module().REVISION
    checkout=sci.ROOT/'research/clojurescript'
    dependency=('{:mvn/version "'+identity+'"}' if kind=='release' else '{:local/root '+json.dumps(str(checkout))+'}')
    expected=[os.environ.get('CLOJURE','clojure'),'-Srepro','-Sdeps',
        '{:paths [] :deps {org.clojure/clojurescript '+dependency+'}}','-M','tests/reference/execution.clj',kind,identity,
        '-' if kind=='release' else str(checkout),str(execution),'cases',str(execution/'cases.tsv')]
    proof.same(command,expected,'Wrong complete original pin compiler command/dependency/settings')
    return [programs['clojure'],*command[1:9],str(output),'program',str(execution/'src'),'semantic.reference']


def checked_location_compile(execution,output,kind,programs):
    command=location_command(execution,output,kind,programs)
    before=source_tree(execution/'src')
    built=sci.report_module().invoke(command,output/'build',timeout=600)
    if built['status']!='completed':raise ValueError('Location-preserving independent pin compiler replay failed')
    identity=kind+' '+(sci.report_module().RELEASE if kind=='release' else sci.report_module().REVISION)
    if ('REFERENCE IDENTITY '+identity not in built['stdout'] or
        'REFERENCE SETTINGS {:optimizations :none, :static-fns false, :def-emits-var false}' not in built['stdout']):
        raise ValueError('Unchecked location-preserving pin compiler identity/settings')
    proof.same(source_tree(execution/'src'),before,'Validated original source tree changed during independent compile')
    return command


def exact_node_replay(unit_root,index,row,programs,*,write=False):
    folder=unit_root/('node-'+str(index));script=node_script(unit_root,index)
    if write:
        ran=sci.report_module().invoke([programs['node'],'-'],folder,source=script,timeout=60)
        if ran['status']!='completed':raise ValueError('Independent location-preserving pin Node execution failed')
        stdout,stderr=ran['stdout'],ran['stderr']
    else:
        command=json.loads((folder/'invocation.json').read_text())
        # Original generic oracle uses its effective command spelling. The
        # independently recorded replay uses the exact resolved executable.
        proof.same(command[1:],['-'],'Changed exact original reference Node command')
        proof.same(str(Path(command[0]).resolve()) if Path(command[0]).is_absolute() else str(Path(__import__('shutil').which(command[0]) or '').resolve()),programs['node'],'Changed actual reference Node executable')
        proof.same((folder/'stdin').read_text(),script,'Changed exact original reference Node input')
        process=subprocess.run([programs['node'],'-'],cwd=sci.ROOT,input=script,capture_output=True,text=True,timeout=60)
        if process.returncode:raise ValueError('Original pin-generated Node replay failed')
        stdout,stderr=process.stdout,process.stderr
        proof.same(stdout,(folder/'stdout').read_text(),'Original pin raw packet differs from executable replay')
        proof.same(stderr,(folder/'stderr').read_text(),'Original pin source markers differ from executable replay')
    packet=sci.report_module().observations(stdout,[row])[row['id']].get('observed')
    proof.same(packet,row['canonical_expected'],'Independent pin exact typed literal differs')
    return {'stdout':stdout,'stderr':stderr,'packet':packet}


def reference_replay(root,document,*,create=False):
    rows=load();before=snapshot();programs=pin_inputs.executable_paths()
    recorded=root.parent/'independent-pin-replay';binding=recorded/'binding.json'
    if create:recorded.mkdir(parents=True,exist_ok=False)
    else:
        stored=json.loads(binding.read_text())
        proof.same(stored['producer_source_sha256'],before,'Independent pin replay producer drift')
        proof.same(stored['reference_execution_binding'],candidates.reference_execution_binding(),'Independent pin replay executable drift')
        manifest={str(p.relative_to(recorded)):sci.digest(p.read_bytes()) for p in sorted(recorded.rglob('*')) if p.is_file() and p!=binding}
        proof.same(stored['artifact_sha256'],manifest,'Independent pin replay artifact census drift')
    packets={};commands={};sources={};units={}
    with tempfile.TemporaryDirectory(prefix='light-original-sci-after191-pin-replay-') as temporary:
        replay=Path(temporary)
        with ThreadPoolExecutor(max_workers=2) as pool:
            futures={kind:pool.submit(sci.reference_cases,candidates.load(),kind,sci.ROOT/'research/clojurescript',replay/kind,600) for kind in ('release','checkout')}
            fresh_layers={kind:future.result() for kind,future in futures.items()}
        for kind in ('release','checkout'):
            execution=root/kind/'execution';generated=replay/kind/'execution';independent=recorded/kind
            proof.same((replay/kind/'preflight/analysis.tsv').read_bytes(),(root/kind/'preflight/analysis.tsv').read_bytes(),'Original pin analysis differs from independent compiler replay')
            original_tree=source_tree(execution/'src');regenerated=source_tree(generated/'src')
            proof.same(regenerated,original_tree,'Original pin complete source differs from compiler replay')
            proof.same((generated/'cases.tsv').read_bytes(),(execution/'cases.tsv').read_bytes(),'Original pin complete source differs from compiler replay')
            for row in rows:proof.same(fresh_layers[kind][row['id']]['observed'],row['canonical_expected'],'Regenerated pin typed packet differs from frozen literal')
            if create:
                for name,data in regenerated.items():
                    destination=independent/'source-tree'/name;destination.parent.mkdir(parents=True,exist_ok=True);destination.write_bytes(data)
                commands[kind]=checked_location_compile(execution,independent,kind,programs)
            else:
                proof.same(source_tree(independent/'source-tree'),regenerated,'Recorded independent complete source tree differs from regeneration')
                commands[kind]=location_command(execution,independent,kind,programs)
                proof.same(json.loads((independent/'build/invocation.json').read_text()),commands[kind],'Changed independently recorded compiler command')
            # The whole source tree was independently reconstructed above. Use
            # the validated original physical input path to preserve exact Var
            # :file metadata, while independently compiling into a fresh tree.
            # No source/metadata/unit/outcome normalization or hash rebinding.
            rebuilt=replay/(kind+'-same-source-location')
            checked_location_compile(execution,rebuilt,kind,programs)
            proof.same(compiled_pin_units(rebuilt),compiled_pin_units(execution),'Original pin complete emitted units differ from compiler replay')
            proof.same(compiled_pin_units(independent),compiled_pin_units(rebuilt),'Independently recorded pin emitted units differ from compiler replay')
            sources[kind]={name:sci.digest(data) for name,data in regenerated.items()}
            units[kind]=compiled_pin_units(independent);packets[kind]={}
            for index,row in enumerate(rows):
                original=exact_node_replay(execution,index,row,programs)
                saved=exact_node_replay(independent,index,row,programs,write=create)
                fresh=exact_node_replay(rebuilt,index,row,programs,write=True)
                proof.same(fresh,original,'Fresh rebuilt pin complete raw packet/source markers differ')
                proof.same(saved,original,'Recorded independent pin complete raw packet/source markers differ')
                packets[kind][row['id']]=saved
            proof.same(source_tree(execution/'src'),original_tree,'Original source tree changed during independent replay')
    proof.same(before,snapshot(),'Original producer/pin inputs changed during executable replay')
    receipt={'schema_version':1,'profile':PROFILE,'scope':'Independent whole-source reconstruction and original-location pin compiler/raw replay; no completion credit',
             'producer_source_sha256':before,'reference_execution_binding':candidates.reference_execution_binding(),
             'source_tree_sha256':sources,'emitted_units_sha256':units,'compiler_commands':commands,'raw_typed_packets':packets,
             'artifact_sha256':{str(p.relative_to(recorded)):sci.digest(p.read_bytes()) for p in sorted(recorded.rglob('*')) if p.is_file() and p!=binding}}
    if create:binding.write_text(json.dumps(receipt,indent=2)+'\n')
    else:proof.same(receipt,stored,'Independent recorded pin replay binding differs from fresh source/compiler/raw replay')
    return receipt

def verified_document(reference_path,phase_root):
    before=snapshot();phase_root=Path(phase_root).resolve()
    document=proof.verified_document(reference_path,phase_root,cohort=sys.modules[__name__])
    proof.same(document['reference_evidence'].get('reference_execution_binding'),candidates.reference_execution_binding(),'Actual whole-source reference executable path/hash drift')
    for row in load():
        frames=document['frames'][row['id']]
        for frame in frames.values():
            proof.same(frame['ordinary_projection'],row['canonical_expected'],'Complete exact whole-source result/effect packet differs')
            proof.same(frame['status'],'passed','Whole-source whole source phase failed')
    executable.executable_replay(phase_root,document)
    document['independent_pin_replay_binding']=reference_replay(Path(reference_path).parent,document)
    proof.same(before,snapshot(),'Whole-source producer changed during complete source replay')
    return document


def reconcile(document):
    proof.same(document.get('profile'),PROFILE,'Wrong whole-source profile')
    proof.same(document.get('compiler_build_producer_source_sha256'),snapshot(),'Current whole-source source/compiler/producer drift')
    reference=Path(document['reference_artifacts_root'])/'evidence.json';phase=Path(document['phase_artifacts_root'])
    replay=verified_document(reference,phase)
    proof.same(document,replay,'Whole-source supplied receipt differs from actual source/executable replay')
    result={}
    for row in load():
        identity=row['id']
        result[row['obligation_id']]={'id':identity,'source_binding':copy.deepcopy(row),
          'layers':copy.deepcopy(replay['reference_evidence']['evidence'][identity]),
          'actual_phase_frames':copy.deepcopy(replay['frames'][identity]),
          'status':'version-difference','reconciliation':'verified-canonical-sci-difference',
          'original_expectation_status':'unresolved','original_semantic_pass_credit':0,
          'canonical_contract_status':'passed','actual_macro_contract_status':'passed',
          'checked_pin_contract_status':{'release':'passed','checkout':'passed'},'focused_owner':None}
    return result


def selected_integration(document):
    """Exercise the real ledger seam; this selected audit cannot grant full credit."""
    import full_ledger as ledger
    rows=load();identities=[row['obligation_id'] for row in rows]
    layers=copy.deepcopy(document['reference_evidence']['evidence'])
    historical={'counts':dict.fromkeys(('covered','missing','version-difference'),0),
                'strict_complete':False,'boundary_strict_complete':True}
    _,witnesses,_=sci.report_module().load_contract('intended-language')
    def selected_rows(result):
        selected=[row for row in result['assertions'] if row['id'] in identities]
        proof.same(len(selected),len(rows),'Whole-source selected ledger cohort missing or duplicated')
        proof.same({row['id'] for row in selected},set(identities),'Whole-source selected ledger identities changed')
        return selected
    absent=ledger.expand_report(historical,witnesses,{},identities=identities,sci_evidence=layers)
    absent_rows=selected_rows(absent)
    proof.same({row['id']:row['status'] for row in absent_rows},
               dict.fromkeys(identities,'missing'),'Whole-source originals credited without whole-source sidecar')
    present=ledger.expand_report(historical,witnesses,{},identities=identities,
                                sci_evidence=layers,original_sci_after191_context_evidence=document)
    present_rows=selected_rows(present)
    proof.same({row['id']:row['status'] for row in present_rows},
               dict.fromkeys(identities,'version-difference'),'Selected whole-source ledger cohort incomplete or credited incorrectly')
    summaries=[]
    for row in present_rows:
        context=row['gap_sci_context_reconciliation'];macro=row['columns']['quickjs_macro']
        proof.same(context['original_semantic_pass_credit'],0,'Whole-source ledger granted original SCI credit')
        proof.same(context['original_expectation_status'],'unresolved','Whole-source ledger changed original SCI expectation')
        proof.same(macro['status'],'passed','Whole-source ledger lacks actual macro proof')
        proof.same({frame['variant'] for frame in macro['frames']},
                   {'macro-fresh/QuickJS','macro-fresh/Node','macro-reused/QuickJS','macro-reused/Node'},
                   'Whole-source selected ledger macro lifecycle frames missing')
        proof.same(len(context['actual_phase_frames']),8,'Whole-source selected ledger lost original application/macro frames')
        proof.same(context['layers'],layers[context['id']],'Whole-source selected ledger layers changed')
        summaries.append({'id':row['id'],'status':row['status'],'original_semantic_pass_credit':0,
                          'original_expectation_status':'unresolved','macro':copy.deepcopy(macro),
                          'actual_phase_frames':copy.deepcopy(context['actual_phase_frames'])})
    for result in (absent,present):
        proof.same(result['selection']['full_scope_selected'],False,'Selected whole-source integration claimed full scope')
        proof.same(result['strict_complete'],False,'Selected whole-source integration claimed full completion')
        proof.same(result['source_strict_complete'],False,'Selected whole-source integration claimed full source completion')
    detached=copy.deepcopy(layers);detached[rows[0]['id']]['release']={}
    try:
        ledger.expand_report(historical,witnesses,{},identities=identities,
                             sci_evidence=detached,original_sci_after191_context_evidence=document)
    except ValueError as error:
        if 'context detached from fresh report execution' not in str(error):raise
        detached_reason=str(error)
    else:raise ValueError('Whole-source selected ledger accepted detached proof layers')
    return {'schema_version':1,'profile':PROFILE,'scope':'selected integration only; no full gate credit',
            'compiler_build_producer_source_sha256':snapshot(),
            'reference_artifacts_root':document['reference_artifacts_root'],
            'phase_artifacts_root':document['phase_artifacts_root'],
            'selection':present['selection'],'counts':{'covered':0,'missing':0,'version-difference':len(rows)},'strict_complete':False,
            'full_report_rows_retained':len(present['assertions']),
            'original_semantic_pass_credit':0,'absent_sidecar_statuses':{row['id']:row['status'] for row in absent_rows},
            'detached_proof_layers_rejected':detached_reason,'assertions':summaries}

    root=Path(directory).resolve();root.mkdir(parents=True,exist_ok=False);before=snapshot()

    (root/'evidence.json').write_text(json.dumps(document,indent=2)+'\n');reconcile(document)
    print('Verified10 unchanged complete SCI originals/80 actualframes/bothpins;zero originalSCIcredit',flush=True);return document
